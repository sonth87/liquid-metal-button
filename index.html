4. Vanilla JS Instance

Khởi tạo và điều khiển trạng thái bằng JavaScript qua LiquidButtonCore.

JavaScript 
import { LiquidButtonCore } from './src/core/LiquidButtonCore.js';
// Hoặc dùng window.LiquidButtonCore nếu dùng bản standalone

const container = document.getElementById('button-slot');
const button = new LiquidButtonCore(container, {
  content: `
    <svg width="16" height="16" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2">
      <path d="M5 12h14M12 5l7 7-7 7"/>
    </svg>
    <span>Continue</span>
  `,
  height: 52,
  borderRadius: "12px",
  paddingX: 28,
  gap: 8,
  textColor: "#ffffff",
  borderColor: "rgba(255, 255, 255, 0.2)",
  btnBg: "#161920",
  fluidColor1: "#ff007a",
  fluidColor2: "#4a00e0"
});

// Lắng nghe click
button.btn.addEventListener('click', () => {
  console.log('Button clicked');
});
